# Optional but recommended: Kraken segmentation (neural blla + classical)
#
# Without Kraken, the GUIs, the web UI and batch_processing.py fall back to
# HPP (horizontal projection profile) segmentation, which is fast but can miss
# most lines on complex pages.
#
# Install Kraken TOGETHER with the other requirements, in a FRESH virtual
# environment, in a single pip call:
#
#   python3 -m venv htr_env && source htr_env/bin/activate
#   pip install -r requirements.txt -r requirements-kraken.txt
#   # GPU: pip install -r requirements-gpu.txt -r requirements.txt -r requirements-kraken.txt \
#   #          --extra-index-url https://download.pytorch.org/whl/cu128
#
# Do NOT add Kraken to an existing environment afterwards. Kraken pins several
# shared packages tightly (safetensors, click, scikit-learn, torch); installed
# in a second step, pip downgrades them without re-checking transformers and
# huggingface_hub, which then fail to import. Resolved in one call, pip picks
# versions that fit everything.
#
# Pinned to the version tested with this repository.

kraken==7.1.1
